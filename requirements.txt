# Kept deliberately minimal per Task #3's design principle: favor lightweight
# dependencies given the Mac Mini's age (2012 model). aiohttp alone covers
# HTTP server, WebSocket server, AND outbound WebSocket client - no need for
# a second library for any of this.
aiohttp==3.10.11

# Task #6 exception to the above: Streamlabs' real-time Socket API uses the
# Socket.IO PROTOCOL (its own handshake/framing on top of WebSocket), not
# plain WebSocket - hand-rolling that correctly risks a silent connection
# failure for a feature where "donations quietly stop granting points" is a
# genuinely bad outcome. Pinned to 4.6.1 deliberately, NOT the latest 5.x -
# verified against python-socketio's own version compatibility chart that
# Streamlabs' server (based on their documented client version) speaks
# Engine.IO protocol revision 3, which requires python-socketio 4.x. The
# newer 5.x targets Engine.IO revision 4 and will not connect correctly.
python-socketio==4.6.1

# Task #8's next-round-credit detection - the pytesseract Python wrapper
# needs the NATIVE tesseract binary installed separately on the Mac Mini
# itself (not something pip can install): `brew install tesseract`.
pytesseract==0.3.13
# Pinned to 12.3.0, NOT an older release - genuinely confirmed the hard
# way: Pillow only started publishing pre-built wheels for Python 3.14
# starting with 12.0.0 (Oct 2025). An earlier pin like 11.0.0 predates
# Python 3.14's existence entirely, so pip falls back to building from
# source, which then fails on missing native jpeg headers. Verified this
# specific version+platform combination has a real, correctly-tagged
# wheel available (pillow-12.3.0-cp314-cp314-macosx_10_15_x86_64.whl) by
# actually downloading it, not just checking release notes.
Pillow==12.3.0
